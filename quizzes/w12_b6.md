    import numpy as np

    def dcg_at_k(rel, k):
        rel = np.asarray(rel[:k], dtype=float)
        return float(np.sum(rel / np.log2(np.arange(1, len(rel) + 1))))

15. `[B6.4]` Roughly what does this print, and why do the two models disagree for $t = 150$ and $t = 300$?

    ```python
    import lightgbm as lgb
    import numpy as np
    from sklearn.linear_model import LinearRegression

    t = np.arange(100, dtype=float).reshape(-1, 1)
    y = 2 * t.ravel() + 5
    gbm = lgb.LGBMRegressor(n_estimators=200, verbose=-1, random_state=0).fit(t, y)
    lin = LinearRegression().fit(t, y)
    X_new = np.array([[50.0], [150.0], [300.0]])
    print(np.round(gbm.predict(X_new), 1), np.round(lin.predict(X_new), 1))
    ```
